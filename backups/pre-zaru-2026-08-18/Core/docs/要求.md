#pragma pack(push, 1)  // 强制字节对齐
typedef struct {
    uint8_t  header[2];      // 0xAA, 0x55
    uint8_t  type;           // 0x01
    uint16_t payload_len;    // 固定为 72
    uint32_t timestamp;
    int32_t  enc_raw[4];
    int16_t  accel_raw[3];
    int16_t  gyro_raw[3];
    float    pose[3];
    float    twist[3];
    float    wheel_speed[4];
    uint8_t  crc8;
} FullFeedbackFrame_t;
#pragma pack(pop)

// 发送函数
void send_full_feedback(void) {
    FullFeedbackFrame_t frame;
    frame.header[0] = 0xAA;
    frame.header[1] = 0x55;
    frame.type = 0x01;
    frame.payload_len = 72;
    frame.timestamp = HAL_GetTick();
    
    // 填充编码器原始值（假设有全局变量）
    frame.enc_raw[0] = encoder_pulse[MOTOR1];
    // ... 填充其他
    
    // 填充IMU原始值（直接从传感器寄存器读取）
    frame.accel_raw[0] = imu.ax; // int16
    // ... 
    
    // 填充融合数据（来自EKF输出）
    frame.pose[0] = ekf_x;
    frame.pose[1] = ekf_y;
    frame.pose[2] = ekf_yaw;
    // ... 填充 twist 和 wheel_speed
    
    // 计算CRC8（包含类型、长度、数据区，不包含帧头和CRC自身）
    frame.crc8 = crc8_calc((uint8_t*)&frame.type, sizeof(frame) - 4); 
    
    // 通过串口DMA或阻塞发送
    HAL_UART_Transmit(&huart1, (uint8_t*)&frame, sizeof(frame), HAL_MAX_DELAY);
}
